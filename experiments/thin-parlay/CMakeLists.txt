cmake_minimum_required(VERSION 3.25)
project(thin_parlay_adapters LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED YES)
set(PARLAYHASH_SOURCE_DIR "" CACHE PATH "ParlayHash checkout with policy constructors and ForEach")
if(NOT EXISTS "${PARLAYHASH_SOURCE_DIR}/include/parlay_hash/unordered_map.h")
  message(FATAL_ERROR "Set PARLAYHASH_SOURCE_DIR to the feature/stateful-hash-policies checkout; released 0.3.0 does not provide the needed constructors or traversal")
endif()
set(SANITIZER "none" CACHE STRING "none, address,undefined, or thread")
find_package(Threads REQUIRED)
enable_testing()
add_executable(thin_adapters test.cpp)
target_include_directories(thin_adapters PRIVATE "${PARLAYHASH_SOURCE_DIR}/include")
target_link_libraries(thin_adapters PRIVATE Threads::Threads)
target_compile_options(thin_adapters PRIVATE -O1 -g -fno-omit-frame-pointer)
if(NOT SANITIZER STREQUAL "none")
 target_compile_options(thin_adapters PRIVATE -fsanitize=${SANITIZER})
 target_link_options(thin_adapters PRIVATE -fsanitize=${SANITIZER})
endif()
add_test(NAME thin_adapters COMMAND thin_adapters)
set_tests_properties(thin_adapters PROPERTIES TIMEOUT 120 ENVIRONMENT "PARLAY_NUM_THREADS=4;ASAN_OPTIONS=halt_on_error=1;TSAN_OPTIONS=halt_on_error=1;UBSAN_OPTIONS=halt_on_error=1")
